> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Adopt the user-supplied hive-themed artwork as the app icon by alpha-masking its pure-black surroundings to transparency and feeding it through the existing, unchanged icon pipeline.

**Architecture:** A one-off scratch Swift script measures the plate's rounded-rect silhouette in the source PNG and re-draws the image through that clip into an RGBA canvas — artwork pixels untouched, corners transparent. The result replaces `assets/app-icon.png` (also the README hero); `scripts/make-icns.sh` regenerates the committed `Resources/AppIcon.icns`; packaging derives the notification catalog from that as it already does.

**Tech Stack:** Swift/AppKit one-off scripts (same toolchain as `make-icns.sh`), sips/iconutil via existing scripts.

**Spec:** `docs/superpowers/specs/2026-07-07-hive-app-icon-design.md`

- [ ] **Step 1: Write the mask script** to `<scratchpad>/mask-icon.swift`:

```swift
import AppKit

let src = CommandLine.arguments[1], dst = CommandLine.arguments[2]
guard let data = FileManager.default.contents(atPath: src),
      let img = NSBitmapImageRep(data: data) else { fatalError("cannot read \(src)") }
let W = img.pixelsWide, H = img.pixelsHigh

func isInk(_ x: Int, _ y: Int) -> Bool {
    guard x >= 0, y >= 0, x < W, y < H, let c = img.colorAt(x: x, y: y) else { return false }
    let t = 8.0 / 255.0
    return c.redComponent > t || c.greenComponent > t || c.blueComponent > t
}

// Plate bounds in top-left pixel coordinates.
var minX = W, maxX = -1, minY = H, maxY = -1
for y in 0..<H { for x in 0..<W where isInk(x, y) {
    if x < minX { minX = x }; if x > maxX { maxX = x }
    if y < minY { minY = y }; if y > maxY { maxY = y }
} }
precondition(maxX > minX && maxY > minY, "no plate found")

// Corner radius: walk down from the plate's top row until the left edge
// has receded to the plate's overall left bound.
var radius = 0
for dy in 0..<((maxY - minY) / 2) {
    let y = minY + dy
    var edge = minX
    while edge <= maxX && !isInk(edge, y) { edge += 1 }
    if edge <= minX + 1 { radius = dy; break }
}
precondition(radius > 0, "no rounded corner detected")

// Re-draw the source through the rounded-rect clip (flip to bottom-left origin).
let plateW = maxX - minX + 1, plateH = maxY - minY + 1
let rectBL = NSRect(x: Double(minX), y: Double(H - minY - plateH),
                    width: Double(plateW), height: Double(plateH))
let out = NSBitmapImageRep(bitmapDataPlanes: nil, pixelsWide: W, pixelsHigh: H,
                           bitsPerSample: 8, samplesPerPixel: 4, hasAlpha: true,
                           isPlanar: false, colorSpaceName: .deviceRGB,
                           bytesPerRow: 0, bitsPerPixel: 0)!
NSGraphicsContext.saveGraphicsState()
NSGraphicsContext.current = NSGraphicsContext(bitmapImageRep: out)
NSBezierPath(roundedRect: rectBL, xRadius: Double(radius), yRadius: Double(radius)).addClip()
img.draw(in: NSRect(x: 0, y: 0, width: Double(W), height: Double(H)))
NSGraphicsContext.restoreGraphicsState()
try! out.representation(using: .png, properties: [:])!.write(to: URL(fileURLWithPath: dst))
print("plate=(\(minX),\(minY)) \(plateW)x\(plateH) radius=\(radius)")
```

- [ ] **Step 3: Write the acceptance check** to `<scratchpad>/verify-mask.swift`:

```swift
import AppKit
let masked = NSBitmapImageRep(data: FileManager.default.contents(atPath: CommandLine.arguments[1])!)!
let source = NSBitmapImageRep(data: FileManager.default.contents(atPath: CommandLine.arguments[2])!)!
// Corners (and a diagonal point outside the plate) must be fully transparent.
for (x, y) in [(2, 2), (1021, 2), (2, 1021), (1021, 1021), (20, 20)] {
    let a = masked.colorAt(x: x, y: y)!.alphaComponent
    precondition(a == 0, "(\(x),\(y)) alpha=\(a), expected 0")
}
// Interior must be byte-identical to the source (alpha exactly 1).
for (x, y) in [(512, 512), (512, 200), (200, 512), (512, 900), (830, 512)] {
    let m = masked.colorAt(x: x, y: y)!, s = source.colorAt(x: x, y: y)!
    precondition(m.alphaComponent == 1, "(\(x),\(y)) interior alpha != 1")
    precondition(abs(m.redComponent - s.redComponent) < 0.005
              && abs(m.greenComponent - s.greenComponent) < 0.005
              && abs(m.blueComponent - s.blueComponent) < 0.005,
                 "(\(x),\(y)) interior color differs from source")
}
print("mask OK")
```
